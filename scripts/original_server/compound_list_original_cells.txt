import numpy as np

import torch
import torch.nn as nn
import copy

from torch.utils.data import TensorDataset, DataLoader
from sklearn.model_selection import train_test_split

from scipy.stats import pearsonr


# ============================================================

device = torch.device("cpu")

# CPU thread도 과도하게 쓰지 않도록 제한
torch.set_num_threads(2)

print("Device:", device)
print("PyTorch CPU threads:", torch.get_num_threads())


# ============================================================
# MLP
# A/B 모두 같은 구조
# input dimension만 다름
# ============================================================

class CompareMLP(nn.Module):

    def __init__(self, input_dim):

        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(input_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 128),
            nn.ReLU(),
            nn.Linear(128, 109)
        )

    def forward(self, x):
        return self.net(x)


# ============================================================
# 모델 하나 학습하는 함수
# ============================================================

def train_model(
    X_train,
    Y_train,
    X_val,
    Y_val,
    seed
):

    torch.manual_seed(seed)
    np.random.seed(seed)

    Xtr = torch.tensor(X_train, dtype=torch.float32)
    Ytr = torch.tensor(Y_train, dtype=torch.float32)

    Xva = torch.tensor(X_val, dtype=torch.float32)
    Yva = torch.tensor(Y_val, dtype=torch.float32)

    # generator를 따로 고정해서
    # A/B의 batch shuffle도 최대한 동일하게 맞춤
    generator = torch.Generator()
    generator.manual_seed(seed)

    loader = DataLoader(
        TensorDataset(Xtr, Ytr),
        batch_size=32,
        shuffle=True,
        generator=generator,
        num_workers=0
    )

    model = CompareMLP(
        input_dim=X_train.shape[1]
    ).to(device)

    criterion = nn.MSELoss()

    optimizer = torch.optim.Adam(
        model.parameters(),
        lr=1e-3
    )

    best_val = float("inf")
    best_state = None

    patience = 30
    patience_count = 0

    for epoch in range(500):

        # ---------------- TRAIN ----------------
        model.train()

        for xb, yb in loader:

            xb = xb.to(device)
            yb = yb.to(device)

            optimizer.zero_grad()

            pred = model(xb)

            loss = criterion(pred, yb)

            loss.backward()
            optimizer.step()


        # ---------------- VALIDATION ----------------
        model.eval()

        with torch.no_grad():

            val_pred = model(
                Xva.to(device)
            )

            val_loss = criterion(
                val_pred,
                Yva.to(device)
            ).item()


        if val_loss < best_val:

            best_val = val_loss

            best_state = copy.deepcopy(
                model.state_dict()
            )

            patience_count = 0

        else:

            patience_count += 1


        if patience_count >= patience:
            break


    model.load_state_dict(best_state)

    return model, best_val

X_struct_raw = data[
    structure_cols
].to_numpy(dtype=np.float32)

    Y_vivo_cols
].to_numpy(dtype=np.float32)

    sorted(data["compound"].unique())
)

    print(f"\n===== SEED {seed} =====")


    # ========================================================
    # 1. 동일한 compound-level split
    # ========================================================

    train_comp, temp_comp = train_test_split(
        all_compounds,
        test_size=0.30,
        random_state=seed
    )

    val_comp, test_comp = train_test_split(
        temp_comp,
        test_size=0.50,
        random_state=seed
    )


    train_m = data["compound"].isin(train_comp).to_numpy()
    val_m   = data["compound"].isin(val_comp).to_numpy()
    test_m  = data["compound"].isin(test_comp).to_numpy()


    # ========================================================
    # 2. Structure scaler
    # train에서만 fit
    # ========================================================

    struct_scaler = StandardScaler()

    Xtr_s = struct_scaler.fit_transform(
        X_struct_raw[train_m]
    )

    Xva_s = struct_scaler.transform(
        X_struct_raw[val_m]
    )

    Xte_s = struct_scaler.transform(
        X_struct_raw[test_m]
    )


    # ========================================================

            torch.tensor(
                Xte_s,

        "test_compounds": len(test_comp),

        "test_conditions": int(test_m.sum()),

        "S_RMSE": rmse_S,

        "S_r": r_S
    })


    print(
        f"Structure only: "
        f"RMSE={rmse_S:.4f}, "
        f"r={r_S:.4f}"
    )

    structure_cols

# 2. Dose one-hot
#
# Low    = [1,0,0]
# Middle = [0,1,0]
# High   = [0,0,1]
# ============================================================

dose_onehot = pd.get_dummies(
    data["dose"]
)

# 항상 동일한 순서로

    columns=["Low", "Middle", "High"],
    fill_value=0
)

X_dose = dose_onehot.to_numpy(
    dtype=np.float32
)

print("\nDose columns:", list(dose_onehot.columns))

    "Total input dimensions:",

)
